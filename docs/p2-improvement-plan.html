P2-1: Python API 端到端测试

与 Node 版的区别

Node 版的 E2E 方案依赖 Playwright（前端浏览器自动化），需要在 trip-front 中编写 TypeScript 代码。

Python 专属方案改为：使用 pytest + httpx.AsyncClient 对后端 API 做 E2E 测试，不依赖前端浏览器，不引入 Node/TypeScript。只需后端运行 + MySQL + Redis（可降级内存）。

目标：覆盖 6 个核心 API 流程的端到端测试，验证完整的请求-响应链路（含数据库读写），在 PR 合并前自动运行。

关键技术决策

选型选择理由
测试框架pytest + pytest-asyncio与现有 642 个测试统一框架，零额外依赖
HTTP 客户端httpx.AsyncClient与现有集成测试一致，支持流式 SSE 读取
后端管理subprocess 启动 uvicorn 子进程脚本 run_all_benchmarks.py 已验证此模式可行
数据库MySQL（复用开发库）E2E 测试应使用真实数据库而非 SQLite
数据隔离前置 fixture 创建专有测试数据，后置清理避免 E2E 测试相互污染

        [PYTHON, "-m", "uvicorn", "src.main:app",

        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,

             "RATE_LIMIT_GLOBAL_MAX": "999999"},
    )

    for i in range(20):
        try:

                r = await c.get("/docs")
            if r.status_code == 200:
                break
        except Exception:
            pass
        time.sleep(1)
    else:
        proc.kill()
        raise RuntimeError("Backend failed to start")

    yield proc
    proc.terminate()
    proc.wait()

"""E2E: 旅行规划核心流程"""
import pytest


@pytest.mark.asyncio
async def test_trip_chat_stream(auth_client):
    """发送旅行规划请求，验证 SSE 流式输出"""
    chunks = []
    async with auth_client.stream(
        "POST", "/api/trip/chat",
        json={"message": "北京 2 天经典游"},
        timeout=120,
    ) as resp:
        assert resp.status_code == 200

        async for line in resp.aiter_lines():
            if line.startswith("data: "):
                import json
                data = json.loads(line[6:])
                if data.get("type") == "chunk":
                    chunks.append(data.get("content", ""))
                elif data.get("type") == "complete":
                    # 验证行程结构
                    assert "北京" in "".join(chunks)
                    assert 2 in [int(s) for s in "".join(chunks) if s.isdigit()]
                    break

    assert len(chunks) > 0, "Should receive at least one chunk"


@pytest.mark.asyncio
async def test_history_after_trip(auth_client):
    """创建行程后，历史记录应包含该行程"""
    # Step 1: 发送规划请求
    await auth_client.post("/api/trip/chat",
                           json={"message": "杭州 1 天"})

    # Step 2: 查询历史
    resp = await auth_client.get("/api/history/trips")
    data = resp.json()
    assert resp.status_code == 200
    assert "data" in data


@pytest.mark.asyncio
async def test_unauthenticated_access(client):
    """未认证用户访问受保护端点应返回 401"""
    resp = await client.get("/api/history/trips")
    assert resp.status_code == 401

"""Eval 报告对比脚本——检查通过率是否退化"""
import json
import sys
from pathlib import Path


def main():
    current_path = sys.argv[sys.argv.index("--current") + 1] if "--current" in sys.argv else "eval-reports/latest.json"
    baseline_path = sys.argv[sys.argv.index("--baseline") + 1] if "--baseline" in sys.argv else "eval-reports/baseline.json"
    threshold = float(sys.argv[sys.argv.index("--threshold") + 1]) if "--threshold" in sys.argv else 0.8
    output = sys.argv[sys.argv.index("--output") + 1] if "--output" in sys.argv else "eval-reports/comparison.json"

    def load(p: str) -> dict:
        fp = Path(p)
        return json.loads(fp.read_text()) if fp.exists() else {}

    current = load(current_path)
    baseline = load(baseline_path)

    diffs = []
    for name, cur_rate in current.get("summary", {}).get("byEvaluator", {}).items():
        prev_rate = baseline.get("summary", {}).get("byEvaluator", {}).get(name, 0)
        change = round((cur_rate - prev_rate) * 100, 1)
        diffs.append({
            "name": name,
            "previous": round(prev_rate * 100, 1),
            "current": round(cur_rate * 100, 1),
            "change": change,
        })

    overall = current.get("summary", {}).get("passRate", 0)
    passed = overall >= threshold

    result = {
        "timestamp": current.get("timestamp", ""),
        "mode": current.get("mode", ""),
        "overallPassRate": round(overall * 100, 1),
        "threshold": round(threshold * 100, 1),
        "passed": passed,
        "diffs": diffs,
    }

    Path(output).parent.mkdir(parents=True, exist_ok=True)
    Path(output).write_text(json.dumps(result, indent=2, ensure_ascii=False))

    if not passed:
        print(f"FAIL: Overall {result['overallPassRate']}% < threshold {result['threshold']}%")
        sys.exit(1)
    print(f"PASS: Overall {result['overallPassRate']}% >= threshold {result['threshold']}%")


if __name__ == "__main__":
    main()

"""性能基线对比——与历史 baseline 对比，检测退化"""
import json
import sys
from pathlib import Path

DATA_DIR = Path(__file__).resolve().parent.parent / "docs" / "performance-data"

# 退化阈值配置
THRESHOLDS = {
    "qps_degradation": 0.15,      # QPS 下降超 15% 告警
    "latency_degradation": 0.15,  # P99 延迟升高超 15% 告警
}


def load_json(name: str) -> dict:
    fp = DATA_DIR / f"{name}.json"
    return json.loads(fp.read_text()) if fp.exists() else {}


def compare_http(baseline: dict, current: dict) -> list[dict]:
    """对比 HTTP 压测结果"""
    regressions = []
    for endpoint in ["login", "history"]:
        base = baseline.get(endpoint, {})
        cur = current.get(endpoint, {})
        if not base or not cur:
            continue

        # QPS (effectiveQps)
        b_qps = base.get("effectiveQps", 0)
        c_qps = cur.get("effectiveQps", 0)
        if b_qps > 0:
            change = (c_qps - b_qps) / b_qps
            if change < -THRESHOLDS["qps_degradation"]:
                regressions.append({
                    "endpoint": f"http/{endpoint}",
                    "metric": "effectiveQps",
                    "baseline": round(b_qps, 3),
                    "current": round(c_qps, 3),
                    "change": round(change * 100, 1),
                })

        # P99 延迟
        b_p99 = base.get("p99", 0)
        c_p99 = cur.get("p99", 0)
        if b_p99 > 0:
            change = (c_p99 - b_p99) / b_p99
            if change > THRESHOLDS["latency_degradation"]:
                regressions.append({
                    "endpoint": f"http/{endpoint}",
                    "metric": "p99",
                    "baseline": round(b_p99, 2),
                    "current": round(c_p99, 2),
                    "change": round(change * 100, 1),
                })

    return regressions


def main():
    # 加载最新结果
    http_current = load_json("http-results")
    sse_current = load_json("sse-results")

    # 加载基线
    baseline = load_json("baseline")

    # 对比
    all_regressions = []
    all_regressions.extend(compare_http(
        baseline.get("http", {}),
        http_current,
    ))

    # 输出
    report = {
        "timestamp": __import__("time").time(),
        "passed": len(all_regressions) == 0,
        "totalChecks": 0,
        "regressions": all_regressions,
    }

    print(json.dumps(report, indent=2))

    if all_regressions:
        print(f"FAIL: {len(all_regressions)} regression(s) detected")
        sys.exit(1)
    print("PASS: All metrics within threshold")


if __name__ == "__main__":
    main()

benchmark_cli.py（统一入口）

"""性能基准测试统一 CLI 入口

用法:
  uv run python scripts/benchmark_cli.py         # 全流程（HTTP + SSE + 图表 + 对比）
  uv run python scripts/benchmark_cli.py http     # 仅 HTTP
  uv run python scripts/benchmark_cli.py --ci     # CI 模式（main 分支更新基线）
"""
import subprocess, sys, os
from pathlib import Path

PROJECT = Path(__file__).resolve().parent.parent
PYTHON = str(PROJECT / ".venv" / "bin" / "python")


def run(script: str, env=None):
    cmd = [PYTHON, script]
    full_env = {**os.environ, **(env or {})}
    result = subprocess.run(cmd, cwd=PROJECT, capture_output=True, text=True)
    print(result.stdout)
    if result.stderr:
        print(result.stderr, file=sys.stderr)
    return result.returncode


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else "all"
    is_ci = "--ci" in sys.argv

    # 压测环境变量（放松限流）
    bench_env = {
        "RATE_LIMIT_AUTH_MAX": "99999",
        "RATE_LIMIT_GLOBAL_MAX": "999999",
        "RATE_LIMIT_CHAT_MAX": "99999",
        "SSE_TOTAL_STREAMS": "5",
        "SSE_CONCURRENCY_LEVELS": "1,5",
    }

    steps = []
    if mode in ("all", "http"):
        steps.append(("scripts/benchmark_http.py", bench_env))
    if mode in ("all", "sse"):
        steps.append(("scripts/benchmark_sse.py", bench_env))
    if mode in ("all", "charts"):
        steps.append(("scripts/benchmark_lib/chart.py", {"MPLCONFIGDIR": "/tmp/mplcfg"}))

    exit_code = 0
    for script, env in steps:
        code = run(script, env)
        if code != 0:
            exit_code = code

    # 对比基线（仅在 full run 或显式指定时）
    if mode == "all" or "--compare" in sys.argv:
        code = run("scripts/benchmark_compare.py")
        if code != 0:
            exit_code = code

    sys.exit(exit_code)


if __name__ == "__main__":
    main()
